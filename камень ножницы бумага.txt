import random

choices = ["камень", "ножницы", "бумага"]
wins_against = {
    "камень": "ножницы",
    "ножницы": "бумага",
    "бумага": "камень",
}

player_score = 0
computer_score = 0

while True:
    player = input("\nВыберите (камень, ножницы, бумага) или 'выход': ").strip().lower()
    if player == "выход":
        break
    if player not in choices:
        print("Неверный ввод. Попробуйте снова.")
        continue

    computer = random.choice(choices)
    print(f"Компьютер выбрал: {computer}")

    if player == computer:
        print("Ничья!")
    elif wins_against[player] == computer:
        print("Вы победили!")
        player_score += 1
    else:
        print("Вы проиграли!")
        computer_score += 1

    print(f"Счёт — Вы: {player_score} | Компьютер: {computer_score}")
