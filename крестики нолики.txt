board = list(range(1, 10))

def draw_board():
    print("-" * 13)
    for i in range(3):
        print(f"| {board[0 + i*3]} | {board[1 + i*3]} | {board[2 + i*3]} |")
        print("-" * 13)

def take_input(player_token):
    while True:
        answer = input(f"Куда поставим {player_token}? ")
        if not answer.isdigit():
            print("Ошибка: введите число от 1 до 9.")
            continue
        
        move = int(answer)
        if 1 <= move <= 9:
            if str(board[move - 1]) not in "XO":
                board[move - 1] = player_token
                break
            else:
                print("Эта клетка уже занята!")
        else:
            print("Некорректный ввод. Введите число от 1 до 9.")

def check_win():
    win_coords = (
        (0, 1, 2), (3, 4, 5), (6, 7, 8), # горизонтали
        (0, 3, 6), (1, 4, 7), (2, 5, 8), # вертикали
        (0, 4, 8), (2, 4, 6)             # диагонали
    )
    for each in win_coords:
        if board[each[0]] == board[each[1]] == board[each[2]]:
            return board[each[0]]
    return False

def main():
    counter = 0
    while True:
        draw_board()
        if counter % 2 == 0:
            take_input("X")
        else:
            take_input("O")
        counter += 1
        
        if counter > 4:
            winner = check_win()
            if winner:
                draw_board()
                print(f"{winner} выиграл!")
                break
        if counter == 9:
            draw_board()
            print("Ничья!")
            break

if __name__ == "__main__":
    main()
